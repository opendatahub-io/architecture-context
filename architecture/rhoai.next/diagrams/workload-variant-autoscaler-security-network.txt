      inferencepools → get, list, watch

Role Bindings:
  - manager-clusterrolebinding → manager-role → SA: controller-manager
  - leader-election-rolebinding → leader-election-role → SA: controller-manager
  - metrics-auth-clusterrolebinding → metrics-auth-role → SA: controller-manager
  - metrics-reader-rolebinding → metrics-reader → SA: controller-manager
  - epp-metrics-reader-clusterrolebinding → epp-metrics-reader-role → SA: epp-metrics-reader

Secrets:
  - epp-metrics-token (kubernetes.io/service-account-token)
    Purpose: Long-lived SA token for EPP metrics reader
    Provisioned: Kubernetes SA token controller
    Auto-Rotate: No

  - controller-manager-token (kubernetes.io/service-account-token)
    Purpose: SA token for ServiceMonitor authentication (OpenShift)
    Provisioned: Kubernetes
    Auto-Rotate: No

  - metrics-server-cert (kubernetes.io/tls)
    Purpose: TLS serving certificate for metrics endpoint (OpenShift)
    Provisioned: OpenShift service-ca operator
    Auto-Rotate: Yes (automatic)

TLS Configuration:
  - MinVersion: TLS 1.2
  - InsecureSkipVerify: configurable (default false on OpenShift, true on base)
  - FIPS: GOEXPERIMENT=strictfipsruntime, CGO_ENABLED=1
  - Certificate handling: Configurable CA cert path, client cert/key paths,
    falls back to system trust store; OpenShift uses service-ca mounted certs
