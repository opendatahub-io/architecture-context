  Controller (workspaces-manager-role):
  - "" : pods, configmaps, events, namespaces → get, list, watch
  - "" : pods/exec → create (activity probing)
  - "" : serviceaccounts, services → full CRUD
  - apps : statefulsets → full CRUD
  - kubeflow.org : workspaces, workspacekinds → full CRUD + finalizers + status
  - networking.istio.io : virtualservices → full CRUD
  - rbac.authorization.k8s.io : clusterroles → bind
  - rbac.authorization.k8s.io : rolebindings → full CRUD
  - gateway.networking.k8s.io : httproutes, referencegrants → full CRUD
  - config.openshift.io : apiservers → get, list, watch

  Backend (workspaces-backend):
  - kubeflow.org : workspaces, workspacekinds → full CRUD
  - "" : namespaces, configmaps → get, list, watch
  - "" : secrets → get, list, watch, create, update, delete
  - "" : pods → get, list, watch; pods/log → get
  - "" : persistentvolumeclaims → get, list, watch, create, delete
  - "" : persistentvolumes → get, list, watch
  - storage.k8s.io : storageclasses → get, list, watch
  - metrics.k8s.io : pods → get, list
  - authorization.k8s.io : subjectaccessreviews → create

  User-Facing (aggregated):
  - kubeflow-workspaces-admin → workspacekinds full CRUD
  - kubeflow-workspaces-edit → workspaces create, delete, patch, update
  - kubeflow-workspaces-view → workspaces get, list, watch

Secrets:
  - workspaces-controller-webhook-cert (kubernetes.io/tls)
    Purpose: TLS cert for webhook server (9443)
    Provisioned: OpenShift service-ca / cert-manager
    Auto-Rotate: Yes (OpenShift auto-rotation)

  - workspaces-controller-metrics-tls (kubernetes.io/tls)
    Purpose: TLS cert for metrics endpoint
    Provisioned: OpenShift service serving cert annotation
    Auto-Rotate: Yes

NetworkPolicies:
  - default-deny ingress in controller namespace
  - Webhook-specific: allows ingress from any source on webhook port only
  - Istio mode: AuthorizationPolicy restricts backend to istio-ingressgateway SA

Authentication Methods:
  - user_token mode: Bearer token → TokenReview → SubjectAccessReview (RHOAI/ODH)
  - internal mode: kubeflow-userid/kubeflow-groups headers (Kubeflow)
  - Multi-header injection defense: rejects duplicate auth headers

TLS Configuration:
  - Controller: OpenShift TLS profile (reads apiservers.config.openshift.io/cluster)
    - Default: Mozilla Intermediate (min TLS 1.2, AEAD ciphers)
    - Supports: Modern (TLS 1.3), Old (TLS 1.0), Custom
    - Profile watcher triggers graceful restart on change
  - Backend: Optional TLS via --cert-file/--key-file; min TLS 1.3 when enabled
  - FIPS: Controller built with GOEXPERIMENT=strictfipsruntime, CGO_ENABLED=1
